   📍 src/com/prestamos/util/ValidacionesNegocio.java

   • Calcular interés generado
   • Calcular multas por mora (2%)
   • Calcular total a pagar

   📍 src/com/prestamos/dao/ArticuloDAO.java

   Métodos implementados:

   • obtenerArticuloPorId() - Buscar artículo específico
   • listarTodosArticulos() - Lista completa
   • listarArticulosDisponibles() - Solo ÓPTIMO/FUNCIONABLE
   • actualizarArticulo() - Modificar datos

   • aplicarMulta() - Aplicar multa por mora

🔐 REGLAS DE NEGOCIO IMPLEMENTADAS
════════════════════════════════════════════════════════════════════════════════

✅ Artículos defectuosos NO se aceptan como garantía
✅ Valor calculado según precio de mercado vigente
✅ Monto máximo del préstamo: 80% del valor tasado
✅ Calificación mínima del cliente: 3.0
✅ Plazo máximo: 12 meses (365 días)
✅ Tasa de interés variable según plazo:
   • Menos de 3 meses → 5%
   • Entre 3 y 6 meses → 10%
   • Más de 6 meses → 15%
✅ Multa por mora: 2% sobre saldo pendiente
✅ Préstamo pasa a "EN_MORA" al aplicar multa
✅ No se pueden eliminar artículos con préstamos activos
✅ Cliente sin artículo NO puede solicitar préstamo
✅ Administrador NO puede ser cliente

💡 EJEMPLO DE USO PROGRAMÁTICO
════════════════════════════════════════════════════════════════════════════════

// CREAR ARTÍCULO
ArticuloDAO articuloDAO = new ArticuloDAO();
Articulo articulo = new Articulo();
articulo.setIdArticulo(articuloDAO.obtenerSiguienteId());
articulo.setTipoArticulo("JOYA");
articulo.setDescripcion("Anillo de oro 18k con diamante");
articulo.setEstado("OPTIMO");
articulo.setValorTasado(5000.0);
articulo.setPrecioMercadoBase(6500.0);
articulo.setPorcentajeTasacion(77.0);
articulo.setFechaAvaluo(new Date());

boolean creado = articuloDAO.insertarArticulo(articulo);

// VALIDAR ANTES DE CREAR PRÉSTAMO
boolean valido = ValidacionesNegocio.validarArticuloParaPrestamo(articulo);
double montoMaximo = ValidacionesNegocio.calcularMontoMaximo(articulo.getValorTasado());

// CREAR PRÉSTAMO
PrestamoDAO prestamoDAO = new PrestamoDAO();
Prestamo prestamo = new Prestamo();
prestamo.setIdPrestamo(prestamoDAO.obtenerSiguienteId());
prestamo.setIdCliente(1);
prestamo.setIdArticulo(articulo.getIdArticulo());
prestamo.setIdAsesor(1);
prestamo.setMonto(4000.0); // 80% de 5000
prestamo.setFechaPrestamo(new Date());

Calendar cal = Calendar.getInstance();
cal.add(Calendar.MONTH, 4);
prestamo.setFechaVencimiento(cal.getTime());

boolean prestamoCreado = prestamoDAO.crearPrestamo(prestamo);
